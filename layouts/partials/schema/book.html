"0" -}} {{- $review := dict "@type" "Review" "reviewRating" (dict "@type" "Rating" "ratingValue" . "bestRating" "5" "worstRating" "1" ) "author" (dict "@type" "Person" "name" ($.Site.Params.author.name | default "Harper Reed")) -}} {{- $book = merge $book (dict "review" $review) -}} {{- end -}} {{- end -}} {{- /* goodreads_link maps to a sameAs URL for the work */}} {{- with .Params.goodreads_link -}} {{- $book = merge $book (dict "sameAs" .) -}} {{- end -}} {{- end -}}
