{{/* RSS feed: emit the page-level feed when available, otherwise the site root feed. On the homepage both resolve to the same URL so we emit only one. Attribute values interpolate through the template engine so titles with quotes stay inert. */}} {{- $rootFeed := "" }} {{- with .Site.GetPage "/" }} {{- with .OutputFormats.Get "rss" }} {{- $rootFeed = .Permalink }} {{- end }} {{- end }} {{- $pageFeed := "" }} {{- with .Page }} {{- with .OutputFormats.Get "rss" }} {{- $pageFeed = .Permalink }} {{- end }} {{- end }} {{- if and $pageFeed (ne $pageFeed $rootFeed) }} {{- end }} {{- with $rootFeed }} {{- end }}
